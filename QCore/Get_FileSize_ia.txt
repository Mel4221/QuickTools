using System;
using System.IO;

namespace QuickTools.QCore
{
    public partial class Get
    {
        /// <summary>
        /// Size types for file size representation.
        /// </summary>
        public enum SizeType
        {
            /// <summary>
            /// The gb.
            /// </summary>
            GB,
            /// <summary>
            /// The mb.
            /// </summary>
            MB,
            /// <summary>
            /// The kb.
            /// </summary>
            KB,
            /// <summary>
            /// The b.
            /// </summary>
            B,
            /// <summary>
            /// The int convertible.
            /// </summary>
            IntConvertible
        }

        /// <summary>
        /// Gets the file size in the specified unit.
        /// </summary>
        /// <param name="fileName">File name.</param>
        /// <param name="sizeType">Size type.</param>
        /// <returns>The size of the file as a string.</returns>
        public static string  FileSize(string fileName, SizeType sizeType)
        {
            if (!File.Exists(fileName))
            {
                throw new FileNotFoundException($"File not found at the given path: {fileName}");
            }

            long fileSizeBytes;
            using (var fileStream = new FileStream(fileName, FileMode.Open, FileAccess.Read))
            {
                fileSizeBytes = fileStream.Length;
            }

            return ConvertSize(fileSizeBytes, sizeType);
        }


        /// <summary>
        /// Gets the length of the file in bytes.
        /// </summary>
        /// <param name="fileName">File name.</param>
        /// <returns>The length of the file in bytes.</returns>
        public static long FileLength(string fileName)
        {
            if (!File.Exists(fileName))
            {
                throw new FileNotFoundException($"File not found at the given path: {fileName}");
            }
            return new FileInfo(fileName).Length;
        }
        /// <summary>
        /// Files the size.
        /// </summary>
        /// <returns>The size.</returns>
        /// <param name="file">File.</param>
        public static string FileSize(string file) => null;
        /// <summary>
        /// Converts the given size in bytes to a human-readable format.
        /// </summary>
        /// <param name="length">Length in bytes.</param>
        /// <returns>Human-readable file size.</returns>
        public static string  FileSize(long length)
        {
            return ConvertSize(length, SizeType.B); // Default to bytes
        }

        /// <summary>
        /// Gets the file size based on a byte array.
        /// </summary>
        /// <param name="buffer">Buffer containing file data.</param>
        /// <returns>Human-readable file size.</returns>
        public static string  FileSize(byte[] buffer)
        {
            return ConvertSize(buffer.Length, SizeType.B); // Default to bytes
        }

        /// <summary>
        /// Gets the total file size of multiple files.
        /// </summary>
        /// <param name="files">Array of file names.</param>
        /// <returns>Human-readable total size of the files.</returns>
        public static string GetTotalFileSize(string[] files)
        {
            long totalSize = 0;
            foreach (var file in files)
            {
                if (File.Exists(file))
                {
                    totalSize += new FileInfo(file).Length;
                }
            }
            return  FileSize(totalSize);
        }

        /// <summary>
        /// Converts bytes to a specified size type.
        /// </summary>
        /// <param name="length">Length in bytes.</param>
        /// <param name="sizeType">Desired size type.</param>
        /// <returns>Formatted size string.</returns>
        private static string ConvertSize(long length, SizeType sizeType)
        {
            switch (sizeType)
            {
                case SizeType.GB:
                    return $"{length / (1024 * 1024 * 1024)} GB";
                case SizeType.MB:
                    return $"{length / (1024 * 1024)} MB";
                case SizeType.KB:
                    return $"{length / 1024} KB";
                case SizeType.IntConvertible:
                    return length.ToString();
                default:
                    return $"{length} B";
            }
        }

        /// <summary>
        /// Gets or sets the delay for human-readable output.
        /// </summary>
        public static int HumanDelay { get; set; } = 500;

      
        //public static bool AllowDebugger { get; set; } = false;
    }
}
